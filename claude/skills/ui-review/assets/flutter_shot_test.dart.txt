// Temporary screenshot test. Copy to test/zz_shot_test.dart, replace every TODO, run, delete. Never commit it.
// Run: FLUTTER_ROOT_DIR=<flutter root> SHOT_DIR=<folder> SHOT_TAG=after flutter test test/zz_shot_test.dart
// TODO: import the app's theme, providers and the screen.
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:go_router/go_router.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  final out = Platform.environment['SHOT_DIR']!;
  final tag = Platform.environment['SHOT_TAG']!;
  setUpAll(() async {
    // TODO: list the app's fonts from pubspec.yaml.
    final fonts = {'ChakraPetch': ['Medium', 'SemiBold', 'Bold'], 'Inter': ['Regular', 'Medium', 'SemiBold']};
    for (final e in fonts.entries) {
      final l = FontLoader(e.key);
      for (final w in e.value) { l.addFont(rootBundle.load('assets/fonts/${e.key}-$w.ttf')); }
      await l.load();
    }
    for (final f in Directory('assets/fonts').listSync().whereType<File>()) {
      if (f.path.contains('JetBrainsMono') && f.path.endsWith('.ttf')) {
        final l = FontLoader('JetBrainsMono')..addFont(rootBundle.load(f.path));
        await l.load();
      }
    }
    final b = File('${Platform.environment['FLUTTER_ROOT_DIR']}/bin/cache/artifacts/material_fonts/MaterialIcons-Regular.otf').readAsBytesSync();
    final i = FontLoader('MaterialIcons')..addFont(Future.value(ByteData.view(b.buffer)));
    await i.load();
  });
  for (final br in Brightness.values) {
    for (final w in [375.0, 320.0]) {
      testWidgets('saved ${br.name} $w', (t) async {
        t.view.physicalSize = Size(w, w == 375 ? 812 : 568);
        t.view.devicePixelRatio = 1;
        addTearDown(t.view.reset);
        SharedPreferences.setMockInitialValues({});
        final c = ProviderContainer();
        addTearDown(c.dispose);
        // TODO: wait for the data the screen needs. Example from Demonica:
        await t.runAsync(() => c.read(gameDataProvider.future));
        await t.pumpWidget(UncontrolledProviderScope(
          container: c,
          child: RepaintBoundary(
            key: const Key('shot'),
            child: MaterialApp.router(
              debugShowCheckedModeBanner: false,
              // TODO: the app's theme, the screen, and any route it opens. Example from Demonica:
              theme: demonicaTheme(br),
              routerConfig: GoRouter(routes: [
                GoRoute(path: '/', builder: (_, _) => const SavedScreen()),
                GoRoute(path: '/smt4/demons', builder: (_, _) => const SizedBox()),
              ]),
            ),
          ),
        ));
        await t.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 200)));
        await t.pump();
        final ex = t.takeException();
        print('$tag ${br.name} $w exception: $ex');
        await t.runAsync(() async {
          final o = t.renderObject<RenderRepaintBoundary>(find.byKey(const Key('shot')));
          final img = await o.toImage(pixelRatio: 2);
          final d = await img.toByteData(format: ui.ImageByteFormat.png);
          File('$out/$tag-${br.name}-${w.toInt()}.png').writeAsBytesSync(d!.buffer.asUint8List());
        });
      });
    }
  }
}
